TM_01_ST_Licensee Win Loss
    # Define test data
    ${tm_fromdate}=    Get Time    \    NOW - 7d
    ${tm_todate}=    Get Time    \    NOW - 1d
    ${tm_filtered_licensee}=    Set Variable    12Bet
    ${tm_clicked_row}=    Set Variable    1
    # 1. Navigate to Win Loss Report > Licensee Win Loss
    navigate menu    Win Loss Report    Licensee Win Loss
    # 2. Select filters then click Submit button
    submit filters for licensee win loss summary    All    Date Range    ${tm_fromdate}    ${tm_todate}    All    EUR
    ...    All    All    All    Normal    No
    take applications screenshot    LA_Pro    TM_01_WinLossReport_LicenseeWinLoss_Summary.jpg
    # Special Case: If "No Information Available" text displays, we'll pass this test case since there is no data
    ${ac_no_info_status}=    Run Keyword And Return Status    Page Should Contain    No Information Available
    Run Keyword If    '${ac_no_info_status}'=='True'    Pass Execution    There is no report data at this moment.
    # VP. Check report table
    Element Should Be Visible    ${report_table_reports}
    # VP. check export excel
    check export excel function    LicenseeWinLoss.*
    # Get licensee name for checking later
    ${tm_licensee_drilldown}=    Get Text    xpath=//*[@id='mainGrid']/tbody/tr[${tm_clicked_row}]/td[@class='lalign party wauto']/a
    # 3. Click 1st licensee link in report
    click licensee link on win loss report    ${tm_clicked_row}
    # VP. Check details report table
    Element Should Be Visible    ${report_table_reports}
    # VP. Clicked licensee should displayed on Licensee listbox filter
    Element Should Be Visible    xpath=//*[@id='party-filter']/div/div[@class='control-header' and contains(text(),'${tm_licensee_drilldown}')]
    # 4. Click Back button
    Click Element    ${back_button}
    wait page loading
    # 5. Click Details link
    Click Element    ${details_link}
    # 6. Select filters then click Submit button
    submit filters for licensee win loss details    ${tm_filtered_licensee}    Date Range    ${tm_fromdate}    ${tm_todate}    All    None
    ...    All    All    All    No Draw    No
    take applications screenshot    LA_Local    TM_01_WinLossReport_LicenseeWinLoss_Detail.jpg
    # Special Case: If "No Information Available" text displays, we'll pass this test case since there is no data
    ${ac_no_info_status}=    Run Keyword And Return Status    Page Should Contain    No Information Available
    Run Keyword If    '${ac_no_info_status}'=='True'    Pass Execution    There is no report data at this moment.
    # VP. check export excel
