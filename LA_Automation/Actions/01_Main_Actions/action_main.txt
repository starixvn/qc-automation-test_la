select date picker
    [Arguments]    ${date type}    ${date}    # ${date type} = From OR To
    [Documentation]    This action is used to select a specific date of calendar date picker.
    ...
    ...    Example:
    ...
    ...    select date picker | From | 2015-03-17
    ...
    ...    select date picker | To | 2015-03-18
    # split specific date
    ${yyyy1}=    Get Slice From List    ${date}    0    4
    ${mm1}=    Get Slice From List    ${date}    5    7
    ${mm11}=    Get Slice From List    ${date}    5    7
    ${dd1}=    Get Slice From List    ${date}    8    10
    # click date picker
    Run Keyword If    '${date type}' == 'From'    click element on report    ${from_date_picker}
    Run Keyword If    '${date type}' == 'To'    click element on report    ${to_date_picker}
    # click Today
    Run Keyword If    '${date type}' == 'From'    click element on report    ${from_date_picker_today}
    Run Keyword If    '${date type}' == 'To'    click element on report    ${to_date_picker_today}
    # select year in date picker
    # get year displayed on date picker
    ${yyyy}=    Get Text    xpath=//div[@id='fd-${date type}Date']${date_picker_year}
    ${yyyy}=    Convert To Integer    ${yyyy}
    @{ITEMS}    Create List    ${yyyy}    ${yyyy - 1}    ${yyyy - 2}    ${yyyy - 3}    ${yyyy - 4}
    ...    ${yyyy - 5}
    : FOR    ${ELEMENT}    IN    @{ITEMS}
    \    Run Keyword If    '${ELEMENT}' != '${yyyy1}'    Click Element    xpath=//div[@id='fd-${date type}Date']${date_picker_previous_year}
    \    Run Keyword If    '${ELEMENT}' == '${yyyy1}'    Exit For Loop
    #get month displayed on date picker
    ${mm}=    Get Text    xpath=//div[@id='fd-${date type}Date']${date_picker_month}
    ${mm}=    Set Variable If    '${mm}'=='JANUARY${SPACE}'    1    '${mm}'=='FEBRUARY${SPACE}'    2    '${mm}'=='MARCH${SPACE}'
    ...    3    '${mm}'=='APRIL${SPACE}'    4    '${mm}'=='MAY${SPACE}'    5    '${mm}'=='JUNE${SPACE}'
    ...    6    '${mm}'=='JULY${SPACE}'    7    '${mm}'=='AUGUST${SPACE}'    8    '${mm}'=='SEPTEMBER${SPACE}'
    ...    9    '${mm}'=='OCTOBER${SPACE}'    10    '${mm}'=='NOVEMBER${SPACE}'    11    '${mm}'=='DECEMBER${SPACE}'
    ...    12
    # Compare value between current month and user input month
    ${mm1}=    Convert To Integer    ${mm1}
    ${difference}=    Evaluate    int(${mm}) - ${mm1}
    ${status}=    Evaluate    ${difference} > 0
    ${steps}=    Evaluate    abs(${difference})
    # select month in date picker
    : FOR    ${INDEX}    IN RANGE    1    12
    \    Run Keyword If    ${steps}==0    Exit For Loop
    \    Run Keyword If    '${status}' == 'True'    Click Element    xpath=//div[@id='fd-${date type}Date']${date_picker_previous_month}
    \    Run Keyword If    '${status}' != 'True'    Click Element    xpath=//div[@id='fd-${date type}Date']${date_picker_next_month}
    \    Run Keyword If    '${INDEX}' == '${steps}'    Exit For Loop
    # select day in date picker
    Click Element    xpath=//div[@id='fd-${date type}Date']//td[contains(@class,'yyyymm-${yyyy1}${mm11} mmdd-${mm11}${dd1}')]

select date range by month
    [Arguments]    ${year}    ${month}    ${from date}    ${to date}
    [Documentation]    This action is used to select date range in Month-From-To calendar.
    # Handle month index: add 0 before number of month if month is less than 10 and greater than 0
    ${ac_temp_zero_month}    Catenate    SEPARATOR=    0    ${month}
    ${ac_month}    Set Variable IF    1 <= int(${month}) <= 9    ${ac_temp_zero_month}
    # Click on Month dropdow list
    Click Element    xpath=//div[@class='monthPicker']/span
    # Check specified year is selected or not. If it is not selected, click on it
    ${ac_attribute}=    Get Element Attribute    xpath=//div[@class='monthPickerDropDown']//a[@year='${year}']@class
    ${ac_year_selected_status}    Run Keyword And Return Status    Should Contain    ${ac_attribute}    selected
    Run Keyword If    '${ac_year_selected_status}'=='False'    Click Element    xpath=//div[@class='monthPickerDropDown']//a[@year='${year}']
    # Click on month
    Run Keyword If    1 <= int(${month}) <= 9    Click Element    xpath=//li[@class='month']/a[@year='${year}' and @month='${ac_month}' ]
    Run Keyword If    10 <= int(${month})    Click Element    xpath=//li[@class='month']/a[@year='${year}' and @month='${month}' ]
    # Select from date and to date
    Select From List By Value    id=ddlFromDay    ${from date}
    Select From List By Value    id=ddlToDay    ${to date}

wait for export file
    [Arguments]    ${filepath}
    [Documentation]    This action is used to wait for a export file.
    : FOR    ${ac_index}    IN RANGE    0    45
    \    ${ac_exist}    Run Keyword And Return Status    File Should Exist    ${filepath}
    \    ${ac_saving}    Run Keyword And Return Status    File Should Exist    ${filepath}.part
    \    Run Keyword If    '${ac_exist}'!='True' or '${ac_saving}'=='True'    sleep    2s
    \    Exit For Loop If    '${ac_index}'==45 or ('${ac_exist}'=='True' and '${ac_saving}'!='True')
